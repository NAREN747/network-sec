! ------------------------------------------------------------
! ACL-INTERNET-IN — Inbound on Gi0/1 (WAN → Router)

 permit tcp any host 10.10.20.5 eq 80     remark LB HTTP
 permit tcp any host 10.10.20.5 eq 443    remark LB HTTPS
 permit tcp any host 10.10.20.10 eq 80    remark Web1 HTTP
 permit tcp any host 10.10.20.10 eq 443   remark Web1 HTTPS
 permit tcp any host 10.10.20.11 eq 80    remark Web2 HTTP
 permit tcp any host 10.10.20.11 eq 443   remark Web2 HTTPS

 remark --- PERMIT: Mail to DMZ (if exposed) ---

 permit tcp any host 10.10.20.25 eq 587   remark Mail Relay Submission

 permit tcp any any established
 permit udp any any eq 53
 remark --- PERMIT: ICMP Echo Reply (for troubleshooting) ---
 permit icmp any any echo-reply
 permit icmp any any time-exceeded
 permit icmp any any unreachable
 remark --- IMPLICIT DENY ALL ---
! Applied: interface Gi0/1 → ip access-group ACL-INTERNET-IN in
